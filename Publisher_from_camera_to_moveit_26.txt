#!/usr/bin/env python3

import rclpy, time
from rclpy.node import Node
from geometry_msgs.msg import PoseStamped


class DualArmTestPublisher(Node):
    def __init__(self):
        super().__init__("dual_arm_test_publisher")

        # Publishers
        self.pub_right = self.create_publisher(
            PoseStamped, "/right_arm/target_pose", 10
        )
        self.pub_left = self.create_publisher(
            PoseStamped, "/left_arm/target_pose", 10
        )

        self.toggle = False  # switch between arms

        self.get_logger().info("Test publisher started")
        self.publish_targets()
        time.sleep(10)
        self.publish_targets()

    def create_pose(self, x, y, z, ox, oy, oz, ow):
        msg = PoseStamped()
        msg.header.frame_id = "camera_link"  # ⚠️ must match your robot
        msg.header.stamp = self.get_clock().now().to_msg()

        msg.pose.position.x = x
        msg.pose.position.y = y
        msg.pose.position.z = z

        msg.pose.orientation.x = ox
        msg.pose.orientation.y = oy
        msg.pose.orientation.z = oz
        msg.pose.orientation.w = ow

        return msg

    def publish_targets(self):
        if self.toggle:
            # RIGHT arm target (Using YOUR numbers)
            msg = self.create_pose(0.308567, 0.30987, 0.218000, 0.0, 0.0, 0.0, 1.0)
            self.pub_right.publish(msg)
            self.get_logger().info("Sent YOUR target to RIGHT arm")
        else:
            # LEFT arm target (Using YOUR numbers)
            msg = self.create_pose(-0.18, 0.252, 0.3123, 0.0, 0.0, 0.0, 1.0)
            self.pub_left.publish(msg)
            self.get_logger().info("Sent YOUR target to LEFT arm")

        self.toggle = not self.toggle


def main():
    rclpy.init()
    node = DualArmTestPublisher()

    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass

    node.destroy_node()
    rclpy.shutdown()


if __name__ == "__main__":
    main()
