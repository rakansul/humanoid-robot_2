GEAR_RATIOS = {
    11: 60.0 / 27.0, 21: 60.0 / 27.0,
    12: 50.0 / 25.0, 13: 50.0 / 25.0, 15: 50.0 / 25.0,
    22: 50.0 / 25.0, 23: 50.0 / 25.0, 25: 50.0 / 25.0,

class Abeed(Node):
    def __init__(self):
        super().__init__("Robot_Driver")

        # Robot State Publisher
        self.rsp = self.create_publisher(
            JointState, "/joint_states", 10
        )

        self.robot = from_json(CONFIG_PATH)
        time.sleep(0.2)

        for m in self.robot.motors:
            m.compliant = True
        time.sleep(0.3)

        timer_period = 0.01
        self.timer = self.create_timer(timer_period, self.publish_states)
        self.right_sub = self.create_subscription( # creat sub
            JointTrajectoryControllerState,
            '/Right_arm_controller/controller_state',
            self.right_arm_callback,
            10)

        self.left_sub = self.create_subscription(
            JointTrajectoryControllerState,
            '/left_arm_controller/controller_state',
            self.left_arm_callback,
            10)

    def right_arm_callback(self, msg):
        self.execute_state_command("right_arm", msg)

    def left_arm_callback(self, msg):
        self.execute_state_command("left_arm", msg)

    def execute_state_command(self, group_name, msg):
        # Extract the target positions MoveIt wants the arm to be at right now
        target_radians = msg.reference.positions
        joint_names = msg.joint_names

        # Convert MoveIt's radians into a dictionary of degrees for your driver
        command_dict = {}
        for name, rad in zip(joint_names, target_radians):
            command_dict[name] = math.degrees(rad)

        # Send the calculated degrees straight to the physical motors
        self.give_command(group_name, command_dict)
    def apply_gear_ratio(self, angle, motor):
        return angle * GEAR_RATIOS.get(motor.id, 1.0)

    def reverse_gear_ratio(self, angle, motor):
        ratio = GEAR_RATIOS.get(motor.id, 1.0)
        return angle / ratio if ratio else angle

    def get_pos(self, group_name):
        group = getattr(self.robot, group_name)
        return {m.name: self.reverse_gear_ratio(m.present_position, m) for m in group}

    def give_command(self, group_name, joint_positions):
        group = getattr(self.robot, group_name)
        for m in group:
            if m.name in joint_positions:
                m.compliant = False
                print(self.apply_gear_ratio(joint_positions[m.name], m))
                m.goal_position = self.apply_gear_ratio(joint_positions[m.name], m)

    def kill_all_torque(self, delay=1.0):
        time.sleep(delay)
        for m in self.robot.motors:
            m.compliant = True
            m.torque_limit = 0
        time.sleep(0.3)

    def publish_states(self):
        msg = JointState()
        msg.header.stamp = self.get_clock().now().to_msg()
        for m in self.robot.motors:
            try:
                msg.name.append(m.name)
                # reverse_gear_ratio gets the actual joint angle in degrees
                pos_deg = self.reverse_gear_ratio(m.present_position, m)
                msg.position.append(math.radians(float(pos_deg)))
            except Exception as e:
                self.get_logger().error("failed to read")
                pass
        self.rsp.publish(msg)
        print(msg)
